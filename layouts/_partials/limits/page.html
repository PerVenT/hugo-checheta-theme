"rule" $rule "path" $key "file" $file) -}} {{- end -}} {{- end -}} {{- end -}} {{- with $params.hero -}} {{- with hugo.Data.limits.hero_title -}} {{- $title := trim (printf "%s %s" ($params.hero.title | default "") ($params.hero.accent | default "")) " " -}} {{- partial "limits/check.html" (dict "v" $title "rule" . "path" "hero.title + hero.accent" "file" $file) -}} {{- end -}} {{- end -}} {{- end -}}
